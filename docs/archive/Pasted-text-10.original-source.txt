package ai.vlink.net

import android.util.Base64
import java.security.SecureRandom
import javax.crypto.Cipher
import javax.crypto.spec.GCMParameterSpec
import javax.crypto.spec.SecretKeySpec

object Crypto {
    private val rnd = SecureRandom()

    fun newKey() = ByteArray(32).also { rnd.nextBytes(it) }

    fun seal(key: ByteArray, plain: ByteArray): ByteArray {
        val nonce = ByteArray(12).also { rnd.nextBytes(it) }
        val c = Cipher.getInstance("AES/GCM/NoPadding")
        c.init(Cipher.ENCRYPT_MODE, SecretKeySpec(key, "AES"), GCMParameterSpec(128, nonce))
        return nonce + c.doFinal(plain)
    }

    fun open(key: ByteArray, blob: ByteArray): ByteArray? {
        if (blob.size < 13) return null
        return runCatching {
            val c = Cipher.getInstance("AES/GCM/NoPadding")
            c.init(Cipher.DECRYPT_MODE, SecretKeySpec(key, "AES"), GCMParameterSpec(128, blob.copyOfRange(0, 12)))
            c.doFinal(blob.copyOfRange(12, blob.size))
        }.getOrNull()
    }

    fun b64(b: ByteArray) = Base64.encodeToString(b, Base64.NO_WRAP)
    fun unb64(s: String) = Base64.decode(s, Base64.NO_WRAP)
}

package ai.vlink.data

import android.content.ContentValues
import android.content.Context
import android.database.sqlite.SQLiteDatabase
import android.database.sqlite.SQLiteOpenHelper
import org.json.JSONObject

class LocalDb(ctx: Context) : SQLiteOpenHelper(ctx, "vlink.db", null, 1) {
    override fun onCreate(d: SQLiteDatabase) {
        d.execSQL("CREATE TABLE devices(id TEXT PRIMARY KEY, name TEXT, model TEXT, sdk INTEGER, online INTEGER DEFAULT 0, lastSeen INTEGER)")
        d.execSQL("CREATE TABLE events(id INTEGER PRIMARY KEY AUTOINCREMENT, deviceId TEXT, kind TEXT, body TEXT, mediaPath TEXT, created INTEGER, seen INTEGER DEFAULT 0)")
    }

    override fun onUpgrade(d: SQLiteDatabase, o: Int, n: Int) {}

    fun addDevice(id: String, name: String, model: String, sdk: Int) {
        val v = ContentValues().apply {
            put("id", id); put("name", name); put("model", model); put("sdk", sdk)
            put("lastSeen", System.currentTimeMillis())
        }
        writableDatabase.insertWithOnConflict("devices", null, v, SQLiteDatabase.CONFLICT_REPLACE)
    }

    fun setOnline(id: String, online: Boolean) {
        writableDatabase.execSQL("UPDATE devices SET online=?, lastSeen=? WHERE id=?",
            arrayOf(if (online) 1 else 0, System.currentTimeMillis(), id))
    }

    fun devices(): List<Array<String>> {
        val out = mutableListOf<Array<String>>()
        readableDatabase.rawQuery("SELECT id,name,model,online FROM devices ORDER BY lastSeen DESC", null).use { c ->
            while (c.moveToNext()) out.add(arrayOf(c.getString(0), c.getString(1) ?: "", c.getString(2) ?: "", c.getString(3)))
        }
        return out
    }

    fun addEvent(deviceId: String, kind: String, body: String, mediaPath: String? = null) {
        val v = ContentValues().apply {
            put("deviceId", deviceId); put("kind", kind); put("body", body)
            put("mediaPath", mediaPath); put("created", System.currentTimeMillis())
        }
        writableDatabase.insert("events", null, v)
    }

    fun events(deviceId: String? = null, limit: Int = 300): List<Array<String?>> {
        val out = mutableListOf<Array<String?>>()
        val sql = if (deviceId == null)
            "SELECT id,deviceId,kind,body,mediaPath,created FROM events ORDER BY id DESC LIMIT $limit"
        else
            "SELECT id,deviceId,kind,body,mediaPath,created FROM events WHERE deviceId=? ORDER BY id DESC LIMIT $limit"
        readableDatabase.rawQuery(sql, deviceId?.let { arrayOf(it) }).use { c ->
            while (c.moveToNext()) out.add(arrayOf(c.getString(0), c.getString(1), c.getString(2), c.getString(3), c.getString(4), c.getString(5)))
        }
        return out
    }

    fun lastLocations(deviceId: String, limit: Int = 400): List<Pair<Double, Double>> {
        val out = mutableListOf<Pair<Double, Double>>()
        readableDatabase.rawQuery(
            "SELECT body FROM events WHERE deviceId=? AND kind='location' ORDER BY id DESC LIMIT $limit",
            arrayOf(deviceId)).use { c ->
            while (c.moveToNext()) runCatching {
                val j = JSONObject(c.getString(0))
                out.add(j.getDouble("lat") to j.getDouble("lon"))
            }
        }
        return out
    }

    fun clear(deviceId: String) {
        writableDatabase.delete("events", "deviceId=?", arrayOf(deviceId))
    }
}

package ai.vlink.net

import android.graphics.Bitmap
import com.google.zxing.BarcodeFormat
import com.google.zxing.qrcode.QRCodeWriter
import org.json.JSONObject

data class PairPayload(val psk: ByteArray, val relay: String, val topic: String) {
    fun encode() = JSONObject().put("v", 1).put("k", Crypto.b64(psk)).put("r", relay).put("t", topic).toString()

    fun qr(size: Int = 800): Bitmap {
        val m = QRCodeWriter().encode(encode(), BarcodeFormat.QR_CODE, size, size)
        val bmp = Bitmap.createBitmap(size, size, Bitmap.Config.ARGB_8888)
        for (x in 0 until size) for (y in 0 until size)
            bmp.setPixel(x, y, if (m[x, y]) 0xFF000000.toInt() else 0xFFFFFFFF.toInt())
        return bmp
    }

    companion object {
        fun decode(s: String) = runCatching {
            val j = JSONObject(s)
            PairPayload(Crypto.unb64(j.getString("k")), j.getString("r"), j.getString("t"))
        }.getOrNull()
    }

class EventSink(private val ctx: Context, private val key: ByteArray, private val db: LocalDb) {

        tmp.renameTo(dst)
        val kind = when {
            mime.startsWith("video") -> "video"
            mime.startsWith("audio") -> "audio"
            else -> "photo"
        }
        deliver(deviceId, kind, JSONObject().put("mime", mime), dst.absolutePath)
    }
}

package ai.vlink.net

import android.content.Context
import fi.iki.elonen.NanoHTTPD
import org.json.JSONObject
import java.io.File
import java.util.concurrent.ConcurrentHashMap
import java.util.concurrent.LinkedBlockingQueue
import java.util.concurrent.TimeUnit

class MasterServer(
    private val ctx: Context,
    private val key: ByteArray,
    private val sink: EventSink,
    private val onPair: (String, String, Int) -> Pair<String, String>?
) : NanoHTTPD(ai.vlink.core.Const.PORT) {

    private val tokens = ConcurrentHashMap<String, String>()
    private val pending = ConcurrentHashMap<String, LinkedBlockingQueue<String>>()

    fun authorize(token: String, deviceId: String) { tokens[token] = deviceId }

    fun push(deviceId: String, cmd: JSONObject) {
        pending.getOrPut(deviceId) { LinkedBlockingQueue() }.offer(cmd.toString())
    }

    private fun read(s: IHTTPSession): ByteArray {
        val n = s.headers["content-length"]?.toIntOrNull() ?: 0
        val buf = ByteArray(n)
        var off = 0
        while (off < n) {
            val r = s.inputStream.read(buf, off, n - off)
            if (r <= 0) break
            off += r
        }
        return buf
    }

    private fun dev(s: IHTTPSession) = s.headers["x-token"]?.let { tokens[it] }

    override fun serve(s: IHTTPSession): Response = try {
        when {
            s.uri == "/pair" && s.method == Method.POST -> {
                val j = Crypto.open(key, read(s))?.let { JSONObject(String(it)) }
                val r = j?.let { onPair(it.optString("name"), it.optString("model"), it.optInt("sdk")) }
                if (r == null) denied() else ok(Crypto.seal(key,
                    JSONObject().put("deviceId", r.first).put("token", r.second).toString().toByteArray()))
            }
            s.uri == "/event" && s.method == Method.POST -> {
                val d = dev(s) ?: return denied()
                val plain = Crypto.open(key, read(s)) ?: return denied()
                sink.ingest(d, plain)
                ok(ByteArray(0))
            }
            s.uri == "/media" && s.method == Method.POST -> {
                val d = dev(s) ?: return denied()
                val f = File(ctx.cacheDir, "up_" + System.currentTimeMillis())
                f.outputStream().use { s.inputStream.copyTo(it) }
                sink.saveUpload(d, f, s.headers["x-mime"] ?: "application/octet-stream")
                ok(ByteArray(0))
            }
            s.uri == "/cmd" && s.method == Method.GET -> {
                val d = dev(s) ?: return denied()
                val wait = s.parameters["wait"]?.firstOrNull()?.toLongOrNull() ?: 25
                val c = pending.getOrPut(d) { LinkedBlockingQueue() }.poll(wait, TimeUnit.SECONDS)
                    ?: return newFixedLengthResponse(Response.Status.NO_CONTENT, "text/plain", "")
                ok(Crypto.seal(key, c.toByteArray()))
            }
            else -> newFixedLengthResponse(Response.Status.NOT_FOUND, "text/plain", "no")
        }
    } catch (e: Exception) {
        newFixedLengthResponse(Response.Status.INTERNAL_ERROR, "text/plain", e.message)
    }

    private fun ok(b: ByteArray) =
        newFixedLengthResponse(Response.Status.OK, "application/octet-stream", b.inputStream(), b.size.toLong())

    private fun denied() = newFixedLengthResponse(Response.Status.UNAUTHORIZED, "text/plain", "auth")
}

object Discovery {
    private fun nsd(c: Context) = c.getSystemService(NsdManager::class.java)

    fun advertise(ctx: Context, id: String) {
        val i = NsdServiceInfo().apply {
            serviceName = "vlink-" + id
            serviceType = ai.vlink.core.Const.NSD_TYPE
            port = ai.vlink.core.Const.PORT
        }
        nsd(ctx).registerService(i, NsdManager.PROTOCOL_DNS_SD, object : NsdManager.RegistrationListener {
            override fun onServiceRegistered(s: NsdServiceInfo) {}
            override fun onRegistrationFailed(s: NsdServiceInfo, e: Int) {}
            override fun onServiceUnregistered(s: NsdServiceInfo) {}
            override fun onUnregistrationFailed(s: NsdServiceInfo, e: Int) {}
        })
    }

    fun browse(ctx: Context, found: (String) -> Unit) {
        val m = nsd(ctx)
        m.discoverServices(ai.vlink.core.Const.NSD_TYPE, NsdManager.PROTOCOL_DNS_SD,
            object : NsdManager.DiscoveryListener {
                override fun onStartDiscoveryFailed(t: String, e: Int) {}
                override fun onStopDiscoveryFailed(t: String, e: Int) {}
                override fun onDiscoveryStarted(t: String) {}
                override fun onDiscoveryStopped(t: String) {}
                override fun onServiceLost(s: NsdServiceInfo) {}
                override fun onServiceFound(s: NsdServiceInfo) {
                    runCatching {
                        m.resolveService(s, object : NsdManager.ResolveListener {
                            override fun onResolveFailed(x: NsdServiceInfo, e: Int) {}
                            override fun onServiceResolved(x: NsdServiceInfo) {
                                x.host?.hostAddress?.let(found)
                            }
                        })
                    }
                }
            })
    }

import android.content.Context
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.withContext
import okhttp3.MediaType.Companion.toMediaTypeOrNull
import okhttp3.OkHttpClient
import okhttp3.Request
import okhttp3.RequestBody.Companion.asRequestBody
import okhttp3.RequestBody.Companion.toRequestBody
import org.json.JSONObject
import java.io.File
import java.util.concurrent.TimeUnit

class Transport(
    private val ctx: Context,
    private val key: ByteArray,
    private val token: String
) {
    @Volatile private var lan: String? = null
    private val http = OkHttpClient.Builder()
        .connectTimeout(8, TimeUnit.SECONDS)
        .readTimeout(35, TimeUnit.SECONDS)
        .build()

    fun lanFound(host: String) { lan = host }

    private suspend fun base(): List<String> = buildList {
        lan?.let { add("http://" + it + ":" + ai.vlink.core.Const.PORT) }
        ai.vlink.core.Config.relay(ctx).ifBlank { null }?.let { add(it.trimEnd('/')) }
    }

    suspend fun send(kind: String, payload: JSONObject): Boolean = withContext(Dispatchers.IO) {
        val body = Crypto.seal(key, JSONObject().put("kind", kind).put("payload", payload).toString().toByteArray())
        for (b in base()) {
            val ok = runCatching {
                http.newCall(Request.Builder().url("$b/event")
                    .header("x-token", token)
                    .post(body.toRequestBody("application/octet-stream".toMediaTypeOrNull()))
                    .build()).execute().use { it.isSuccessful }
            }.getOrDefault(false)
            if (ok) return@withContext true
        }
        false
    }

    suspend fun upload(f: File, mime: String): Boolean = withContext(Dispatchers.IO) {
        for (b in base()) {
            val ok = runCatching {
                http.newCall(Request.Builder().url("$b/media")
                    .header("x-token", token).header("x-mime", mime)
                    .post(f.asRequestBody(mime.toMediaTypeOrNull()))
                    .build()).execute().use { it.isSuccessful }
            }.getOrDefault(false)
            if (ok) return@withContext true
        }
        false
    }

    suspend fun pollCmd(wait: Int): JSONObject? = withContext(Dispatchers.IO) {
        val h = lan ?: return@withContext null
        runCatching {
            http.newCall(Request.Builder()
                .url("http://" + h + ":" + ai.vlink.core.Const.PORT + "/cmd?wait=" + wait)
                .header("x-token", token).get().build()).execute().use { r ->
                if (r.code != 200) null else {
                    val plain = Crypto.open(key, r.body!!.bytes()) ?: return@withContext null
                    JSONObject(String(plain))
                }
            }
        }.getOrNull()
    }

    suspend fun pairOnLan(host: String, name: String, model: String, sdk: Int): Pair<String, String>? =
        withContext(Dispatchers.IO) {
            runCatching {
                val body = Crypto.seal(key, JSONObject().put("name", name).put("model", model).put("sdk", sdk)
                    .toString().toByteArray())
                http.newCall(Request.Builder().url("http://" + host + ":" + ai.vlink.core.Const.PORT + "/pair")
                    .post(body.toRequestBody("application/octet-stream".toMediaTypeOrNull()))
                    .build()).execute().use { r ->
                    if (!r.isSuccessful) return@withContext null
                    val j = JSONObject(String(Crypto.open(key, r.body!!.bytes()) ?: return@withContext null))
                    j.getString("deviceId") to j.getString("token")
                }
            }.getOrNull()
        }
}

package ai.vlink.net

import android.content.Context
import kotlinx.coroutines.CoroutineScope
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.SupervisorJob
import kotlinx.coroutines.delay
import kotlinx.coroutines.isActive
import kotlinx.coroutines.launch
import org.json.JSONObject
import java.io.File

class Outbox(private val ctx: Context, private val key: ByteArray) {
    private val dir = File(ctx.filesDir, "outbox").apply { mkdirs() }
    private val scope = CoroutineScope(SupervisorJob() + Dispatchers.IO)

    fun put(kind: String, payload: JSONObject, media: File? = null, mime: String? = null) {
        val env = JSONObject().put("kind", kind).put("payload", payload)
        if (media != null && media.exists()) {
            env.put("media", media.absolutePath)
            env.put("mime", mime)
        }
        File(dir, System.currentTimeMillis().toString() + "_" + kind + ".bin")
            .writeBytes(Crypto.seal(key, env.toString().toByteArray()))
    }

    fun size() = dir.listFiles()?.size ?: 0

    fun start(t: Transport) {
        scope.launch {
            var backoff = 3_000L
            while (isActive) {
                var progressed = false
                for (f in (dir.listFiles()?.sortedBy { it.name } ?: emptyList())) {
                    val env = Crypto.open(key, f.readBytes())
                        ?.let { runCatching { JSONObject(String(it)) }.getOrNull() }
                    if (env == null) { f.delete(); continue }

                    val mp = env.optString("media").ifBlank { null }
                    if (mp != null) {
                        val mf = File(mp)
                        if (!mf.exists()) { f.delete(); continue }
                        if (t.upload(mf, env.optString("mime", "application/octet-stream"))) {
                            mf.delete(); f.delete(); progressed = true
                        } else break
                    } else {
                        if (t.send(env.getString("kind"), env.getJSONObject("payload"))) {
                            f.delete(); progressed = true
                        } else break
                    }
                }
                backoff = if (progressed) 3_000L else (backoff * 2).coerceAtMost(90_000L)
                delay(backoff)
            }
        }
    }
}

class RelayClient(
    private val relay: String,
    private val masterToken: String,
    private val onBlob: (deviceId: String, blob: ByteArray) -> Unit
) {
    private val http = OkHttpClient.Builder().pingInterval(25, TimeUnit.SECONDS).build()
    private var ws: WebSocket? = null
    @Volatile var alive = false

        val base = relay.trimEnd('/').replaceFirst("https://", "wss://").replaceFirst("http://", "ws://")

            .url(base + "/hub?role=master&token=" + masterToken).build(),
            object : WebSocketListener() {
                override fun onOpen(w: WebSocket, r: Response) { alive = true }
                override fun onFailure(w: WebSocket, t: Throwable, r: Response?) { alive = false }
                override fun onClosed(w: WebSocket, c: Int, r: String) { alive = false }
                override fun onMessage(w: WebSocket, text: String) {
                    val j = runCatching { JSONObject(text) }.getOrNull() ?: return
                    if (j.has("blob")) onBlob(j.optString("src"), Crypto.unb64(j.getString("blob")))
                }
            })

        if (!alive) return false
        return ws?.send(JSONObject().put("to", deviceId).put("blob", Crypto.b64(blob)).toString()) ?: false
    }

    fun close() { alive = false; ws?.close(1000, null); ws = null }
}

import org.eclipse.paho.client.mqttv3.IMqttDeliveryToken

import org.json.JSONObject

class MqttLink(
    private val topic: String,
    private val clientId: String,
    private val isMaster: Boolean,
    private val onBlob: (from: String, blob: ByteArray) -> Unit
) {
    private var client: MqttAsyncClient? = null
    private val sub = "vlink/" + topic + "/" + (if (isMaster) "c2m" else "m2c")
    private val pub = "vlink/" + topic + "/" + (if (isMaster) "m2c" else "c2m")
    @Volatile var alive = false

    fun connect() {
        runCatching {
            val c = MqttAsyncClient(ai.vlink.core.Const.MQTT, clientId, MemoryPersistence())
            c.setCallback(object : MqttCallbackExtended {
                override fun connectComplete(re: Boolean, s: String) {
                    alive = true
                    runCatching { c.subscribe(sub, 1) }
                }
                override fun connectionLost(t: Throwable?) { alive = false }
                override fun deliveryComplete(t: IMqttDeliveryToken?) {}
                override fun messageArrived(t: String, m: MqttMessage) {
                    runCatching {
                        val j = JSONObject(String(m.payload))
                        onBlob(j.optString("from"), Crypto.unb64(j.getString("b")))
                    }
                }
            })
            val o = MqttConnectOptions().apply {
                isAutomaticReconnect = true
                isCleanSession = true
                keepAliveInterval = 30
                connectionTimeout = 15
            }
            c.connect(o).waitForCompletion(12_000)
            c.subscribe(sub, 1)
            client = c
            alive = true
        }
    }

    fun send(to: String, blob: ByteArray): Boolean {
        if (!alive) return false
        return runCatching {
            val p = JSONObject().put("from", clientId).put("b", Crypto.b64(blob)).toString().toByteArray()
            client?.publish(pub, MqttMessage(p).apply { qos = 1 })
            true
        }.getOrDefault(false)
    }

    fun close() { alive = false; runCatching { client?.disconnect() }; client = null }
}
