package ai.vlink.core

import android.app.Notification
import android.app.NotificationChannel
import android.app.NotificationManager
import android.app.PendingIntent
import android.content.Context
import android.content.Intent
import androidx.core.app.NotificationCompat
import ai.vlink.ui.MasterActivity
import org.json.JSONObject

object Notif {
    fun channels(ctx: Context) {
        val m = ctx.getSystemService(NotificationManager::class.java)
        listOf(
            Triple(Const.CH_HUB, "Hub", NotificationManager.IMPORTANCE_LOW),
            Triple(Const.CH_LINK, "Link", NotificationManager.IMPORTANCE_MIN),
            Triple(Const.CH_EVT, "Events", NotificationManager.IMPORTANCE_DEFAULT)
        ).forEach { (id, name, imp) ->
            m.createNotificationChannel(NotificationChannel(id, name, imp).apply { setShowBadge(false) })
        }
    }

    fun ongoing(ctx: Context, ch: String, id: Int, title: String, text: String): Notification {
        val pi = PendingIntent.getActivity(ctx, 0, Intent(ctx, MasterActivity::class.java),
            PendingIntent.FLAG_IMMUTABLE or PendingIntent.FLAG_UPDATE_CURRENT)
        return NotificationCompat.Builder(ctx, ch)
            .setSmallIcon(android.R.drawable.stat_sys_upload)
            .setContentTitle(title).setContentText(text)
            .setOngoing(true).setSilent(true)
            .setPriority(NotificationCompat.PRIORITY_MIN)
            .setContentIntent(pi).build()
    }

    fun event(ctx: Context, kind: String, p: JSONObject) {
        val title = when (kind) {
            "sms" -> "SMS از " + p.optString("from")
            "call" -> "تماس " + p.optString("state") + " - " + p.optString("number")
            "notif" -> p.optString("app") + ": " + p.optString("title")
            "location" -> "موقعیت جدید"
            "photo" -> "عکس جدید"
            "video" -> "ویدیو جدید"
            "audio" -> "صدا جدید"
            "keylog" -> "کیلاگ"
            "file" -> "فایل جدید"
            "ack" -> "تأیید " + p.optString("action")
            else -> kind
        }
        ctx.getSystemService(NotificationManager::class.java).notify(
            System.currentTimeMillis().toInt(),
            NotificationCompat.Builder(ctx, Const.CH_EVT)
                .setSmallIcon(android.R.drawable.ic_dialog_info)
                .setContentTitle(title).setContentText(p.optString("text"))
                .setAutoCancel(true).build()
        )
    }
}
